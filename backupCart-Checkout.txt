export const getProducts = async (params: ProductFilterParams) => {

  const productRepo = AppDataSource.getRepository(Product); // tạo repo để lm vc với bảng product
  const { page = 1, limit = 12, keyword, min_price, max_price, sort, tag_ids, category_id } = params; // destructuring và để mặc định nếu vào trang chủ không chọn gì -> lấy trang đầu hiển thị 12 sp

  const query = productRepo.createQueryBuilder("product")
    .leftJoinAndSelect("product.category", "category")
    .select([
      "product.id", "product.name", "product.basePrice", 
      "product.thumbnailUrl", "product.handle", "product.createdAt",
      "category.name", "category.id"
    ]);

  // filter
  if (keyword) query.andWhere("product.name ILIKE :keyword", { keyword: `%${keyword}%` }); // nếu ng dùng có lọc theo điều kiện này -> thêm where vào câu lệnh sql
  if (min_price) query.andWhere("product.basePrice >= :min", { min: min_price });
  if (max_price) query.andWhere("product.basePrice <= :max", { max: max_price });
  if (category_id) query.andWhere("product.categoryId = :catId", { catId: category_id });
  
  if (tag_ids) {
    const tagIdsArray = tag_ids.split(",").map(id => Number(id));
    query.innerJoin("product.variants", "variant")
         .innerJoin("variant.tags", "tag")
         .andWhere("tag.id IN (:...tags)", { tags: tagIdsArray });
  }

  // sort
  if (sort === 'price_asc') query.orderBy("product.basePrice", "ASC");
  else if (sort === 'price_desc') query.orderBy("product.basePrice", "DESC");
  else query.orderBy("product.createdAt", "DESC");

  // pagination
  const skip = (page - 1) * limit;
  query.skip(skip).take(limit);

  const [data, total] = await query.getManyAndCount();

  return {
    data,
    meta: { total, page, last_page: Math.ceil(total / limit) }
  };
};

import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { jwtVerify } from "jose"; // Dùng thư viện jose cho Edge Runtime

// d/n các Route cần bảo vệ
const privatePaths = [ "/checkout", "/profile", "/user"];
const authPaths = ["/login", "/register"]; // Nếu đã login thì không vào đây nữa
const adminPaths = ["/admin"]; 

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl; // check ng dùng di đâu
  
  // Lấy token từ Cookie
  const accessToken = request.cookies.get("accessToken")?.value;
  
  // chuyển secret key từ chữ -> Uint8Array
  const secret = new TextEncoder().encode(process.env.JWT_SECRET);

  let userRole = null;
  let isAuthenticated = false;

  // Verify Token nếu có
  if (accessToken) {
    try {
      const { payload } = await jwtVerify(accessToken, secret);
      userRole = payload.role; 
      isAuthenticated = true;
    } catch (error) {
      // Token lỗi hoặc hết hạn -> coi như chưa login
      isAuthenticated = false;
    }
  }

  // case 1 - Đã login mà cố vào trang Login/Register -> Đẩy về Home
  // some() -> chạy vòng lặp qua từng phần tử của mảng - Chỉ cần một cái khớp là nó trả về true
  if (isAuthenticated && authPaths.some((path) => pathname.startsWith(path))) {
    return NextResponse.redirect(new URL("/", request.url));
    /*
    - new URL(path, base)
      path bạn muốn chuyển tới: /      
      request.url hiện tại là: http://localhost:3000/login?ref=123
      cách chạy: 
        Lấy Protocol: http
        Lấy Host: localhost:3000
        Thay thế phần đuôi bằng Path mới: /
        Kết quả: http://localhost:3000/
      */
  }

  // case 2 - Truy cập trang Admin
  if (adminPaths.some((path) => pathname.startsWith(path))) {
    if (!isAuthenticated) {
      return NextResponse.redirect(new URL("/login", request.url));
    }
    if (userRole !== "admin") { // Login rồi nhưng không phải admin -> Đẩy về trang chủ 
      return NextResponse.redirect(new URL("/", request.url));
    }
  }

  // case 3 - Truy cập trang Private 
  if (privatePaths.some((path) => pathname.startsWith(path))) {
    if (!isAuthenticated) {
        return NextResponse.redirect(new URL("/login", request.url));
    }
  }

  return NextResponse.next();
}

Đây là một câu hỏi rất hay về kiến trúc hệ thống và tối ưu hiệu suất. Ý tưởng của bạn là chuyển gánh nặng "ghép dữ liệu" (hydration) từ Backend (BE) sang Frontend (FE) để tận dụng khả năng cache của trình duyệt hoặc CDN.

Câu trả lời ngắn gọn là: Có thể nhanh hơn về mặt trải nghiệm người dùng (Perceived Performance), nhưng rủi ro về độ trễ mạng (Network Latency) là rất cao nếu làm không khéo.

Hãy cùng phân tích sâu hơn 2 phương án:

Phương án 1: Cách bạn đang làm (Server-Side Hydration)

BE lấy ID từ Redis -> BE query SQL lấy chi tiết -> BE ghép lại -> Trả về FE cục to.

Ưu điểm:

Chỉ tốn 1 Request: FE chỉ cần gọi GET /cart. Tiết kiệm thời gian thiết lập kết nối (TCP Handshake, SSL).

Tối ưu Database: Bạn đang dùng IN (:...ids) trong SQL. Đây là cách truy vấn tối ưu nhất (Batch Query). Dù giỏ hàng có 100 món, SQL cũng chỉ chạy 1 câu lệnh.

Nhược điểm:

Nếu SQL chậm, toàn bộ API /cart bị chậm. Redis nhanh cũng vô nghĩa vì phải chờ ông SQL.

Response payload (dung lượng phản hồi) có thể lớn.

Phương án 2: Ý tưởng của bạn (Client-Side Hydration + Promise.all)

BE trả về list ID từ Redis (cực nhẹ) -> FE nhận list -> FE gọi Promise.all bắn N request lấy chi tiết từng sản phẩm.

Ưu điểm lý thuyết:

API /cart phản hồi "ngay lập tức" (vì chỉ đọc Redis). App cảm giác rất mượt.

Các API GET /variant/:id có thể được cache ở trình duyệt (HTTP Cache) hoặc CDN. Nếu user F5 lại trang, trình duyệt lấy từ cache đĩa, không cần gọi server nữa.

Nhược điểm chết người (Thực tế):

Vấn đề "Connection Overhead": Nếu giỏ hàng có 10 món, bạn bắn 10 request cùng lúc. Trình duyệt (Chrome/Firefox) thường giới hạn tối đa 6 connection song song đến cùng 1 domain.

-> 4 request còn lại phải xếp hàng chờ (Queueing).

-> Tổng thời gian = Thời gian request chậm nhất + Thời gian chờ.

Rủi ro mạng: Mạng user yếu thì việc gửi 10 request nhỏ sẽ chậm và dễ lỗi hơn gửi 1 request to.

GIẢI PHÁP TỐI ƯU NHẤT (Hybrid Approach)

Thay vì bắt FE chịu khổ (bắn nhiều request) hoặc bắt SQL chịu khổ (query liên tục), hãy dùng Redis để cache luôn thông tin sản phẩm.

Bạn giữ nguyên logic ghép dữ liệu ở Backend, nhưng thay vì ghép từ SQL, bạn ghép từ Redis.

Quy trình đề xuất:

Lấy Cart từ Redis (như cũ).

Lấy danh sách variantId.

Dùng Redis MGET (Multi Get) để lấy thông tin chi tiết sản phẩm (Key ví dụ: product:101, product:102).

Nếu món nào thiếu trong Redis -> Lúc đó mới query SQL và set ngược lại vào Redis (Cache Aside Pattern).

export const getCartService = async (userId: number) => {
  const key = getCartKey(userId);

  // Kết quả trả về dạng Object: { "101": "2", "205": "1" }
  const cartHash = await redis.hgetall(key);

  // Nếu object rỗng -> Trả về giỏ rỗng
  if (!cartHash || Object.keys(cartHash).length === 0) {
    return { id: `redis_${userId}`, userId, items: [] };
  }

  // Chuyển đổi Object của Redis thành Mảng chuẩn: [{ variantId: 101, quantity: 2 }, ...]
  const cartItems = Object.entries(cartHash).map(([vId, qty]) => ({
    variantId: Number(vId), // Key của Hash luôn là string, phải ép kiểu về Number
    quantity: Number(qty),
  }));

  // Hydrate: Lấy thông tin chi tiết sản phẩm từ DB -> để đảm bảo giá/tên luôn mới nhất
  const variantIds = cartItems.map((i) => i.variantId);

  const variants = await variantRepo.createQueryBuilder("variant")
    .leftJoinAndSelect("variant.product", "product")
    .leftJoinAndSelect("variant.images", "images")
    .leftJoinAndSelect("variant.tags", "tags")
    // vc này giúp tốn duy nhất 1 câu truy cấn -> lấy ttin cả giỏ hàng (thay vì chạy vòng for n query từng spham)
    .where("variant.id IN (:...ids)", { ids: variantIds })  // (:...ids) -> sau : báo cho typeOrm là 1 tham số / { ids: variantIds } truyền vào tham số -> tìm kiếm trong các id nằm trong mảng -> chỉ cần 1 câu lệnh sql là đủ 
    .getMany(); // -> thực thi và trả về nh` kq

  //  merge dữ liệu Redis với dữ liệu DB
  const populatedItems = cartItems.map((item) => {
    const variantInfo = variants.find((v) => v.id === item.variantId);// tìm ttin chi tiết của hàng trong giỏ hàng ở bảng chi tiết sp mới lấy từ db
    if (!variantInfo) return null;

    return {
      id: `item_${item.variantId}`, // ID fake -> vì fe cần id mà redis k lưu theo key -> fake id để fe k bị lỗi 
      variantId: item.variantId,
      quantity: item.quantity,
      variant: variantInfo,
    };
  }).filter(Boolean); // loại bỏ null

  return {
    id: `redis_${userId}`,
    userId,
    items: populatedItems,
  };
};
